cmake_minimum_required(VERSION 3.20)
project(eyes_host_tests C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_options(-Wall -Wextra -Werror $<$<COMPILE_LANGUAGE:CXX>:-fno-exceptions>  $<$<COMPILE_LANGUAGE:CXX>:-fno-rtti> -fsanitize=address,undefined -fno-sanitize-recover=undefined)
add_link_options(-fsanitize=address,undefined)
file(GLOB TESTS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/test_*.cpp)
file(GLOB CORE_SRCS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/../../components/*/core/*.cpp)
file(GLOB CAT_SRCS ${CMAKE_CURRENT_SOURCE_DIR}/../../components/cat_eyes/src/*.c)
add_library(cat_eyes STATIC ${CAT_SRCS})
target_compile_options(cat_eyes PRIVATE -w)
target_include_directories(cat_eyes PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/../../components/cat_eyes/include)
target_link_libraries(cat_eyes PUBLIC m)
add_library(cjson STATIC third_party/cJSON.c)
target_compile_options(cjson PRIVATE -w)
add_executable(host_tests main.cpp ${TESTS} ${CORE_SRCS})
target_link_libraries(host_tests PRIVATE cjson cat_eyes)
target_include_directories(host_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/third_party)
target_include_directories(host_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/../../components ${CMAKE_CURRENT_SOURCE_DIR}/../../components/cat_eyes/src)
target_compile_definitions(host_tests PRIVATE GOLDEN_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../golden")
target_compile_definitions(host_tests PRIVATE EFX_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../../../examples/efx")
